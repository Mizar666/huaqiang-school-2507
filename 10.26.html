# ==================== 1. LED 控制类 ====================
class LED:
    """LED 极简控制（默认 4 号引脚），支持开关、翻转、闪烁与调光"""
    def __init__(self, pin=4):
        self.pin_num = pin
        self.io = Pin(pin, Pin.OUT)
        self.pwm = None
        self.state = 0
        self.off()

    def on(self):
        """点亮"""
        self.state = 1
        if self.pwm:
            self.pwm.duty_u16(65535)
        else:
            self.io.value(1)

    def off(self):
        """熄灭"""
        self.state = 0
        if self.pwm:
            self.pwm.duty_u16(0)
        else:
            self.io.value(0)

    def toggle(self):
        """状态翻转"""
        if self.state:
            self.off()
        else:
            self.on()

    def bright(self, level):
        """调光：level 取值 0~1000（0 全暗、1000 最亮）"""
        if self.pwm is None:
            from machine import PWM
            self.pwm = PWM(Pin(self.pin_num), freq=1000, duty_u16=0)
        if level < 0:
            level = 0
        if level > 1000:
            level = 1000
        self.state = 1 if level > 0 else 0
        self.pwm.duty_u16(int(level * 65535 / 1000))

    def blink(self, times=2, delay=0.15):
        """闪烁指定次数"""
        for _ in range(times):
            self.on()
            time.sleep(delay)
            self.off()
            time.sleep(delay)

class Button:
    """按键控制（默认板载 BOOT 键 GPIO 0，按下返回 True）"""
    def __init__(self, pin=0):

class Touch:
    """电容触摸控制（默认 27 号引脚，手指碰触返回 True）"""
    def __init__(self, pin=27):
        from machine import TouchPad
        self.tp = TouchPad(Pin(pin))

    def read(self):
        """读取电容原始读数"""
        return self.tp.read()

    def is_touched(self, threshold=300):
        """判断手指是否触碰"""
        return self.tp.read() < threshold

# 数字简谱：1~7 = 哆来咪发梭拉西，0 或空格 = 停一下
_SCALE = {
    '1': 262, '2': 294, '3': 330, '4': 349,
    '5': 392, '6': 440, '7': 494, '0': 0
}


def _notes_freq(notes):
    """把一串简谱拆成频率表：数字前写 - 是低八度、写 + 是高八度。"""
    out = []
    i = 0
    s = str(notes)
    while i < len(s):
        ch = s[i]
        if ch in "+-" and i + 1 < len(s):
            f = _SCALE.get(s[i + 1], 0)
            out.append(f * 2 if ch == "+" else f // 2)
            i += 2
        else:
            out.append(_SCALE.get(ch, 0))
            i += 1
    return out


# ==================== 4. 蜂鸣器与旋律 ====================
class Buzzer:
    """无源蜂鸣器（默认 23 号引脚）：频率决定音高，必须用 PWM 驱动"""
    def __init__(self, pin=23):
        self.pin_num = pin
        self.pin = Pin(pin, Pin.OUT, value=0)

    def tone(self, freq, duration=0.15):
        """发声：freq = 频率（Hz，越大音越高），duration = 响多久（秒）"""
        if freq <= 0:
            time.sleep(duration)
            return
        from machine import PWM
        p = PWM(self.pin, freq=int(freq), duty=512)   # 512/1023 ≈ 一半，方波最响
        time.sleep(duration)
        p.deinit()                                    # 关掉 PWM
        self.pin.value(0)                             # 引脚拉低，别一直带电

    def beep(self, times=1, delay=0.1):
        """提示音哔哔声"""
        for _ in range(times):
            self.tone(1000, 0.08)
            time.sleep(delay)

    def play(self, notes="1231", beat=0.30):
        """播放数字简谱：1~7 = 哆来咪发梭拉西，0 或空格 = 停一下

        数字前写 - 是低八度、写 + 是高八度（谱子上数字底下/上面的点）
        beat = 每个音符多长（秒）：改大变慢，改小变快
        """
        for f in _notes_freq(notes):
            self.tone(f, beat * 0.8)      # 八成的拍子用来出声
            time.sleep(beat * 0.2)        # 剩下两成留白，两个音才分得开


# ==================== 5. OLED 屏幕类 (防呆容错) ====================
class _SSD1306(framebuf.FrameBuffer):
    def __init__(self, width, height, external_vcc):
        self.width = width
        self.height = height
        self.external_vcc = external_vcc
        self.pages = self.height // 8
        self.buffer = bytearray(self.pages * self.width)
        super().__init__(self.buffer, self.width, self.height, framebuf.MONO_VLSB)

    def show(self):
        self.write_cmd(0x21)
        self.write_cmd(0)
        self.write_cmd(self.width - 1)
        self.write_cmd(0x22)
        self.write_cmd(0)
        self.write_cmd(self.pages - 1)
        self.write_data(self.buffer)

class _SSD1306_I2C(_SSD1306):
    def __init__(self, width, height, i2c, addr=0x3C, external_vcc=False):
        self.i2c = i2c
        self.addr = addr
        self.temp = bytearray(2)
        self.write_list = [b"\x40", None]
        super().__init__(width, height, external_vcc)
        for cmd in (
            0xAE, 0xD5, 0x80, 0xA8, self.height - 1, 0xD3, 0x00,
            0x40, 0x8D, 0x14, 0x20, 0x00, 0xA1, 0xC8, 0xDA,
            0x12 if self.height == 64 else 0x02,
            0x81, 0xCF, 0xD9, 0xF1, 0xDB, 0x40, 0xA4, 0xA6, 0xAF
        ):
            self.write_cmd(cmd)
        self.fill(0)
        self.show()

    def write_cmd(self, cmd):
        self.temp[0] = 0x80
        self.temp[1] = cmd
        self.i2c.writeto(self.addr, self.temp)

    def write_data(self, buf):
        self.write_list[1] = buf
        self.i2c.writevto(self.addr, self.write_list)

class OLED:
    """OLED 屏幕（用法：oled = OLED(16, 17)；没接屏自动忽略、不打断其他实验）"""
    def __init__(self, scl=16, sda=17, width=128, height=64):
        self.ok = False
        self.screen = None
        try:
            self.i2c = SoftI2C(scl=Pin(scl), sda=Pin(sda), freq=100000, timeout=50000)
            devs = self.i2c.scan()
            addr = 0x3C if (0x3C in devs or 60 in devs) else (0x3D if (0x3D in devs or 61 in devs) else None)
            if addr:
                self.screen = _SSD1306_I2C(width, height, self.i2c, addr=addr)
                self.ok = True
                print("[OLED] 屏幕连接成功")
            else:
                print("[OLED 提示] 未检测到屏幕（不影响LED与其他实验运行）")
        except Exception:
            print("[OLED 提示] 屏幕暂未接入")

    def clear(self):
        if self.ok and self.screen:
            try:
                self.screen.fill(0)
                self.screen.show()
            except Exception:
                pass

    def display(self, line1="", line2="", line3="", line4=""):
        """一键全刷屏显示最多 4 行文字"""
        if self.ok and self.screen:
            try:
                self.screen.fill(0)
                for idx, content in enumerate([line1, line2, line3, line4]):
                    if content:
                        self.screen.text(str(content), 0, idx * 16, 1)
                self.screen.show()
            except Exception:
                pass


# ==================== 6. 四位数码管 TM1637 ====================
_CHARS = {
    '0': 0x3F, '1': 0x06, '2': 0x5B, '3': 0x4F, '4': 0x66,
    '5': 0x6D, '6': 0x7D, '7': 0x07, '8': 0x7F, '9': 0x6F,
    'A': 0x77, 'B': 0x7C, 'C': 0x39, 'D': 0x5E, 'E': 0x79, 'F': 0x71,
    'G': 0x3D, 'H': 0x76, 'J': 0x1E, 'L': 0x38, 'N': 0x37, 'O': 0x3F,
    'P': 0x73, 'R': 0x50, 'T': 0x78, 'U': 0x3E, 'Y': 0x6E,
    ' ': 0x00, '-': 0x40, '_': 0x08,
}

COLON_INDEX = 1      # 中间那个冒号挂在第 2 位上；屏上冒号位置不对就改这里


class TM1637:
    def __init__(self, clk=22, dio=21, brightness=5):
        self._delay = 5          # 微秒；屏闪烁或显示乱码就把这个数加大
        self._clk = Pin(clk, Pin.OUT, value=1)
        self._dio = Pin(dio, Pin.OUT, value=1)
        self._brightness = brightness & 0x07
        self._buf = bytearray(4)

    # ---------- 协议层（不用管，想看懂看课件最后一页） ----------
    def _start(self):
        self._clk(1)
        self._dio(1)
        sleep_us(self._delay)
        self._dio(0)
        sleep_us(self._delay)
        self._clk(0)
        sleep_us(self._delay)

    def _stop(self):
        self._clk(0)
        self._dio(0)
        sleep_us(self._delay)
        self._clk(1)
        sleep_us(self._delay)
        self._dio(1)
        sleep_us(self._delay)

    def _write_byte(self, b):
        for _ in range(8):
            self._dio(b & 1)
            sleep_us(self._delay)
            self._clk(1)
            sleep_us(self._delay)
            self._clk(0)
            sleep_us(self._delay)
            b >>= 1
        # 把 DIO 放开收 ACK：从机把它拉低 = 应答
        self._dio.init(Pin.IN, Pin.PULL_UP)
        self._clk(1)
        sleep_us(self._delay)
        ack = not self._dio.value()
        self._clk(0)
        sleep_us(self._delay)
        self._dio.init(Pin.OUT, value=1)
        return ack

    def _send(self, data):
        self._start()
        ok = True
        for b in data:
            if not self._write_byte(b):
                ok = False
        self._stop()
        return ok

    # ---------- 你要用的部分 ----------
    def flush(self):
        """把缓冲区写到屏上；返回 True = 屏应答了。"""
        a = self._send((0x40,))                            # 数据命令：写显存，地址自增
        b = self._send((0xC0,) + tuple(self._buf))          # 从地址 0 写 4 个段码
        c = self._send((0x88 | self._brightness,))          # 显示开 + 亮度
        return a and b and c

    def show(self, text, colon=False):
        """显示文字：最多 4 个字符，靠右对齐；colon=True 时中间冒号亮。"""
        text = str(text)
        buf = bytearray(4)
        i = 4
        for ch in reversed(text[-4:]):
            i -= 1
            buf[i] = _CHARS.get(ch.upper(), 0x00)
        if colon:
            buf[COLON_INDEX] |= 0x80
        self._buf = buf
        return self.flush()

    def number(self, n):
        """显示一个数字，靠右对齐；最大 9999。"""
        n = int(n)
        if n > 9999:
            n = 9999
        if n < 0:
            n = 0
        return self.show(str(n))

    def time(self, hour, minute, colon=True):
        """显示 时:分（自动补零，例：9 点 5 分 → 0905）。"""
        return self.show("%02d%02d" % (int(hour) % 24, int(minute) % 60), colon)

    def mmss(self, secs, running=None):
        """显示 分:秒（最大 99:59）。

        running=True  → 冒号常亮（在走）
        running=False → 冒号一闪一闪（暂停了）
        不传          → 冒号常亮
        """
        secs = int(secs)
        if secs > 5999:
            secs = 5999
        if secs < 0:
            secs = 0
        if running is None:
            colon = True
        elif running:
            colon = True
        else:
            colon = (ticks_ms() // 500) % 2 == 0
        return self.show("%02d%02d" % (secs // 60, secs % 60), colon)

    def clear(self):
        """清空屏幕。"""
        self._buf = bytearray(4)
        return self.flush()

    def brightness(self, level):
        """调亮度：0（最暗）~ 7（最亮）。"""
        self._brightness = level & 0x07
        return self._send((0x88 | self._brightness,))

    def selftest(self):
        """全亮自检：亮一下 8888 + 冒号，返回 True = 接上了。"""
        ok = self.show("8888", colon=True)
        sleep_ms(800)
        return ok

# ==================== 7. 温湿度 DHT11 ====================
class DHT:
    """DHT11 温湿度（默认 13 号引脚）：read() 读一次，temp() 温度，humi() 湿度"""
    def __init__(self, pin=13):
        import dht
        self.io = Pin(pin)
        self.sensor = dht.DHT11(self.io)
        self.t = 0
        self.h = 0

    def read(self):
        """读一次（两次之间至少隔 1 秒，别催它）"""
        self.sensor.measure()
        self.t = self.sensor.temperature()
        self.h = self.sensor.humidity()
        return self.t

    def temp(self):
        """温度（℃），整数"""
        return self.t

    def humi(self):
        """湿度（%）"""
        return self.h

份一 · 外接 LED：一根线串一个电阻

考试里最常考的就是它。32 号口 → 长脚 → 短脚 → 220Ω → GND

怎么接（32 号口）

· 32 号口 → LED 长脚（正极） · LED 短脚 → 220Ω 电阻 · 电阻另一头 → GND · 电阻两脚要横跨两排：插在同一排 = 底下铜片直通，等于没插

两个当场烧东西的坑

· 短脚不串电阻直接怼 3V3 → 当场烧灯 · 长短脚插反 → 不亮，但不会烧，拆了重插就行 · 不亮先看这两条，再看口号写没写错

份一 · 按键：四只脚，只有两对

怎么接（0 号口）

· 一只脚 → 0 号口 · 斜对着那只脚 → GND · 板子内部已经上拉，不用接电阻 · 同侧两只脚 = 本来就是通的，按了也没反应

板子怎么知道你按了

份二 · 数码管：多出来的两根线在"说话"

它跟 LED 不是一类东西 —— LED 给电就亮，数码管要一位一位地把数字送进去

为什么难

· LED 一根线，有电就亮 · 数码管靠 CLK + DIO 两根线传数字 · CLK 打拍子，DIO 趁拍子送 0 和 1 · 两根线谁先谁后不能错 —— 错了它就不吭声

22 / 21 不是规定，是分配

· 板载蓝灯 2 ｜ 按键 0 ｜ 触摸 27 · 外接 LED 32 ｜ 蜂鸣器 23 ｜ 温湿度 13 · OLED 16 / 17 · 数码管 22 / 21 —— 是剩下的两个，不是命定的两个

份二 · 接线卡：四根针，插完先自检

四根针 → 哪儿

· VCC → 3V3（不是 5V！） · GND → GND · CLK → 22 号口 · DIO → 21 号口 · CLK / DIO 接反：电脑不报错，就是黑屏 —— 最坑

想换口？线换、代码也换

· 接 19 / 18 号口，就写 TM1637(19, 18) · 接线换孔，代码里两个数字跟着换，两边必须对得上 · 34–39 号脚只能输入不能输出，接上去永远黑屏

插完先跑一行自检：TM1637().selftest() —— 亮出 8888 就是接对了，黑屏就是没接对。

# 练习：摸一下就亮（答案）
from box import LED, Touch
from time import sleep

led = LED(32)            # 外接灯接 32 号口
tp = Touch(27)           # 触摸线接 27 号口

while True:
    if tp.is_touched():      # 摸到了吗
        led.on()             # 亮
    else:
        led.off()            # 灭
    sleep(0.05)

认错卡：不亮 / 没反应 / 黑屏，先查这几条

别急着换元件，九成是这几条

灯不亮

· 长脚短脚插反了？拆了重插 · 电阻插在同一排了？两脚要跨排 · 代码里口号写错了？接 32 就写 LED(32) · 忘了上电 / 忘了 GND

按键没反应

· 四只脚插在同一侧了 → 换成斜对角两只 · 忘了把另一只脚接 GND · 代码写的口和线上插的口不是一个

摸它没反应

· 捏的是塑料皮，不是金属头 · 线插到了别的口，代码却写 27 · 碰一下要按住一两秒再看

数码管黑屏

· CLK / DIO 接反了（不报错，只黑屏） · 线上插的口 ≠ 代码里的数字（接了 22/21 却写 TM1637(19, 18)） · VCC 接到了 5V

报 ImportError

· box.py 没存进板子上（存到电脑里了） · 文件名要一字不差：box.py

屏太暗

· 默认亮度 5，调到 tm.brightness(7)
