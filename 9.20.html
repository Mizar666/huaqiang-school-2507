# box.py - ESP32 极简教学封装库 (2507班实训版)
import time
from machine import Pin, SoftI2C
import framebuf

class LED:
    """LED 极简控制（默认 4 号引脚）"""
    def __init__(self, pin=4):
        self.io = Pin(pin, Pin.OUT)
        self.off()
    def on(self): self.io.value(1)
    def off(self): self.io.value(0)
    def toggle(self): self.io.value(not self.io.value())
    def blink(self, times=2, delay=0.15):
        for _ in range(times):
            self.on(); time.sleep(delay)
            self.off(); time.sleep(delay)

class Button:
    """按键控制（默认板载 BOOT 键 GPIO 0）"""
    def __init__(self, pin=0):
        self.io = Pin(pin, Pin.IN, Pin.PULL_UP)
    def is_pressed(self): return self.io.value() == 0

class Touch:
    """电容触摸控制（默认 27 号引脚）"""
    def __init__(self, pin=27):
        from machine import TouchPad
        self.tp = TouchPad(Pin(pin))
    def read(self): return self.tp.read()
    def is_touched(self, threshold=300): return self.tp.read() < threshold

class Buzzer:
    """蜂鸣器与旋律播放（默认 23 号引脚）"""
    def __init__(self, pin=23):
        self.pin = Pin(pin, Pin.OUT)
    def tone(self, freq, duration=0.15):
        if freq <= 0: time.sleep(duration); return
        from machine import PWM
        p = PWM(self.pin, freq=int(freq), duty=512)
        time.sleep(duration)
        p.deinit()
    def beep(self, times=1, delay=0.1):
        for _ in range(times):
            self.tone(1000, 0.08); time.sleep(delay)
    def play(self, notes="1231"):
        scale = {'1':262, '2':294, '3':330, '4':349, '5':392, '6':440, '7':494, '0':0}
        for note in str(notes):
            self.tone(scale.get(note, 0), 0.18)
            time.sleep(0.04)

🔌 四步接线清单

· 信号输出：板子 32号口 ➔ 面包板 3b.i
· LED 正极：长脚插入 3b.f（连接32号信号）
· LED 负极：短脚插入 4b.f
· 限流电阻：4b.h 跨接至 负极蓝线轨(-)
· 地线回路：负极蓝线轨 ➔ 板子 GND

from box import LED
import time

# 初始化 32 号接口 LED
led = LED(32)

# 循环闪烁：开灯 0.5 秒，关灯 0.5 秒
while True:
    led.on()         # 点亮黄色小灯
    time.sleep(0.5)  # 保持 0.5 秒
    led.off()        # 熄灭黄色小灯
    time.sleep(0.5)  # 保持 0.5 秒

参考答案 1 · 双灯交替自动闪烁

仅供自查 · 对照连线与逻辑

Python 
from box import LED
import time

# 初始化两个 LED 接口
led1 = LED(32)  # 黄灯：32 号口
led2 = LED(33)  # 红灯：33 号口

# 双灯交替闪烁
while True:
    led1.on()        # 1号灯点亮
    led2.off()       # 2号灯熄灭
    time.sleep(0.5)

    led1.off()       # 1号灯熄灭
    led2.on()        # 2号灯点亮
    time.sleep(0.5)

📋 参考接线对照

· 1号灯（黄）：32号口 ➔ 3b.i ➔ LED ➔ 4b.h 电阻 ➔ 负极轨
· 2号灯（红）：33号口 ➔ 11b.i ➔ LED ➔ 12b.h 电阻 ➔ 负极轨
· 公共地线：负极蓝线轨(-) ➔ 板子 GND
· 核心规律：两个独立控制通道，各自串联电阻，共用负极接地回路

参考答案 2 · 双灯触控互锁切换

仅供自查 · 对照互锁逻辑

Python 
from box import LED, Touch
import time

led1 = LED(32)     # 外接黄灯（32号口）
led2 = LED(33)     # 外接红灯（33号口）
touch = Touch(27)  # 触摸引脚（27号口，默认阈值300）

print("手摸亮灯实验开始！")

while True:
    if touch.is_touched():  # 触摸时返回 True
        led1.on()           # 黄灯亮
        led2.off()          # 红灯灭
    else:                   # 松手未触摸
        led1.off()          # 黄灯灭
        led2.on()           # 红灯亮

    time.sleep(0.05)

📋 互锁逻辑解析

· if 分支：手摸时，强制 led1.on() 并 led2.off()；
· else 分支：松手时，强制 led1.off() 并 led2.on()；
· 无需复杂 sleep 延时，纯由人体触摸实时决定两盏灯的互锁状态！

参考答案 3 · 课堂简答题要点

仅供核对 · 回路与元件原理

📝 简答 1：电流回路流向

· 电源流出：ESP32 32号引脚 输出高电平（3.3V）；
· 进入面包板：绿色跳线导入 3 列（3b.i ➔ 3b.f）；
· 流经小灯：从 LED 长脚（正极） 流入，短脚（负极） 流出至 4 列；
· 限流降压：经 4 列穿过 220Ω 电阻（4b.h ➔ 负极轨）；
· 闭合回到地：沿负极蓝线接地轨回到 ESP32 GND 引脚。

📝 简答 2：电阻的核心作用

· 核心功能：限流保护；
· 原因：LED 灯珠正向导通后自身内阻极小，相当于直通；
· 极端后果：如果不串联电阻，瞬时电流会远超引脚承受极限（ESP32 单引脚上限约 12~20mA）；
· 保护机制：串联 220Ω 电阻将工作电流限制在 安全区间（约 6~8mA），既保障发光，又防止烧毁 LED 与芯片。
