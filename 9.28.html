# box.py - ESP32 极简教学封装库 (2507班实训全功能版)
# 适用平台: MicroPython (ESP32)
# 封装 LED、双灯、按键、触摸引脚、蜂鸣器旋律与 OLED 屏幕

import time
from machine import Pin, SoftI2C
import framebuf

# ==================== 1. LED 控制类 ====================
class LED:
    """LED 极简控制（默认 4 号引脚），支持开关、翻转、闪烁与调光"""
    def __init__(self, pin=4):
        self.pin_num = pin
        self.io = Pin(pin, Pin.OUT)
        self.pwm = None
        self.state = 0
        self.off()

    def on(self):
        """点亮"""
        self.state = 1
        if self.pwm:
            self.pwm.duty_u16(65535)
        else:
            self.io.value(1)

    def off(self):
        """熄灭"""
        self.state = 0
        if self.pwm:
            self.pwm.duty_u16(0)
        else:
            self.io.value(0)

    def toggle(self):
        """状态翻转"""
        if self.state:
            self.off()
        else:
            self.on()

    def bright(self, level):
        """调光：level 取值 0~1000（0 全暗、1000 最亮）"""
        if self.pwm is None:
            from machine import PWM
            self.pwm = PWM(Pin(self.pin_num), freq=1000, duty_u16=0)
        if level < 0:
            level = 0
        if level > 1000:
            level = 1000
        self.state = 1 if level > 0 else 0
        self.pwm.duty_u16(int(level * 65535 / 1000))

    def blink(self, times=2, delay=0.15):
        """闪烁指定次数"""
        for _ in range(times):
            self.on()
            time.sleep(delay)
            self.off()
            time.sleep(delay)

# ==================== 2. 按键控制类 ====================
class Button:
    """按键控制（默认板载 BOOT 键 GPIO 0，按下返回 True）"""
    def __init__(self, pin=0):
        self.io = Pin(pin, Pin.IN, Pin.PULL_UP)

    def is_pressed(self):
        """判断是否按下"""
        return self.io.value() == 0

# ==================== 3. 触摸引脚类 ====================
class Touch:
    """电容触摸控制（默认 27 号引脚，手指碰触返回 True）"""
    def __init__(self, pin=27):
        from machine import TouchPad
        self.tp = TouchPad(Pin(pin))

    def read(self):
        """读取电容原始读数"""
        return self.tp.read()

    def is_touched(self, threshold=300):
        """判断手指是否触碰"""
        return self.tp.read() < threshold

# ==================== 4. 蜂鸣器与旋律 ====================
class Buzzer:
    """无源蜂鸣器（默认 23 号引脚，支持音调与简谱播放）"""
    def __init__(self, pin=23):
        self.pin = Pin(pin, Pin.OUT)

    def tone(self, freq, duration=0.15):
        """发声指定频率（Hz）与时长（秒）"""
        if freq <= 0:
            time.sleep(duration)
            return
        from machine import PWM
        p = PWM(self.pin, freq=int(freq), duty=512)
        time.sleep(duration)
        p.deinit()

    def beep(self, times=1, delay=0.1):
        """提示音哔哔声"""
        for _ in range(times):
            self.tone(1000, 0.08)
            time.sleep(delay)

    def play(self, notes="1231"):
        """播放数字简谱 (1~7 分别代表哆来咪发梭拉西，0为停顿)"""
        scale = {
            '1': 262, '2': 294, '3': 330, '4': 349,
            '5': 392, '6': 440, '7': 494, '0': 0
        }
        for note in str(notes):
            f = scale.get(note, 0)
            self.tone(f, 0.18)
            time.sleep(0.04)

# ==================== 5. OLED 屏幕类 (防呆容错) ====================
class _SSD1306(framebuf.FrameBuffer):
    def __init__(self, width, height, external_vcc):
        self.width = width
        self.height = height
        self.external_vcc = external_vcc
        self.pages = self.height // 8
        self.buffer = bytearray(self.pages * self.width)
        super().__init__(self.buffer, self.width, self.height, framebuf.MONO_VLSB)

    def show(self):
        self.write_cmd(0x21)
        self.write_cmd(0)
        self.write_cmd(self.width - 1)
        self.write_cmd(0x22)
        self.write_cmd(0)
        self.write_cmd(self.pages - 1)
        self.write_data(self.buffer)

class _SSD1306_I2C(_SSD1306):
    def __init__(self, width, height, i2c, addr=0x3C, external_vcc=False):
        self.i2c = i2c
        self.addr = addr
        self.temp = bytearray(2)
        self.write_list = [b"\x40", None]
        super().__init__(width, height, external_vcc)
        for cmd in (
            0xAE, 0xD5, 0x80, 0xA8, self.height - 1, 0xD3, 0x00,
            0x40, 0x8D, 0x14, 0x20, 0x00, 0xA1, 0xC8, 0xDA,
            0x12 if self.height == 64 else 0x02,
            0x81, 0xCF, 0xD9, 0xF1, 0xDB, 0x40, 0xA4, 0xA6, 0xAF
        ):
            self.write_cmd(cmd)
        self.fill(0)
        self.show()

    def write_cmd(self, cmd):
        self.temp[0] = 0x80
        self.temp[1] = cmd
        self.i2c.writeto(self.addr, self.temp)

    def write_data(self, buf):
        self.write_list[1] = buf
        self.i2c.writevto(self.addr, self.write_list)

class OLED:
    """OLED 屏幕（用法：oled = OLED(16, 17)；没接屏自动忽略、不打断其他实验）"""
    def __init__(self, scl=16, sda=17, width=128, height=64):
        self.ok = False
        self.screen = None
        try:
            self.i2c = SoftI2C(scl=Pin(scl), sda=Pin(sda), freq=100000, timeout=50000)
            devs = self.i2c.scan()
            addr = 0x3C if (0x3C in devs or 60 in devs) else (0x3D if (0x3D in devs or 61 in devs) else None)
            if addr:
                self.screen = _SSD1306_I2C(width, height, self.i2c, addr=addr)
                self.ok = True
                print("[OLED] 屏幕连接成功")
            else:
                print("[OLED 提示] 未检测到屏幕（不影响LED与其他实验运行）")
        except Exception:
            print("[OLED 提示] 屏幕暂未接入")

    def clear(self):
        if self.ok and self.screen:
            try:
                self.screen.fill(0)
                self.screen.show()
            except Exception:
                pass

    def display(self, line1="", line2="", line3="", line4=""):
        """一键全刷屏显示最多 4 行文字"""
        if self.ok and self.screen:
            try:
                self.screen.fill(0)
                for idx, content in enumerate([line1, line2, line3, line4]):
                    if content:
                        self.screen.text(str(content), 0, idx * 16, 1)
                self.screen.show()
            except Exception:
                pass

实验 · 让灯呼吸

复制 → 粘贴 → 按 F5

Python 
from box import LED
import time

led = LED(32)          # 32 号口的外接黄灯

while True:
    for n in range(0, 1001, 10):    # 从 0 数到 1000
        led.bright(n)               # 越来越亮
        time.sleep(0.01)
    for n in range(1000, -1, -10):  # 从 1000 数回 0
        led.bright(n)               # 越来越暗
        time.sleep(0.01)

🔍 提示

· 第二盏灯的亮度 = 1000 减去 黄灯的亮度

【规律】填一填

跑通之后再填，慢一点没关系

【真相】灯怎么会「半亮」

一个接口只有两种状态

· 输出高：灯亮
· 输出低：灯灭
· 没有「半亮」这种电平

那就让它开关得飞快

· 一秒钟开关 1000 次
· 开的时间占得少 → 看着就暗
· 眼睛跟不上这么快，只看到变暗

【扩展】掀开箱子：不用库怎么写

一模一样的效果，把外壳拆掉

Python 
from machine import Pin, PWM
import time

led = PWM(Pin(32), freq=1000)   # 32 号口，每秒开关 1000 次

while True:
    for n in range(0, 1001, 10):
        led.duty_u16(n * 65)     # 真正认的范围是 0 ~ 65535
        time.sleep(0.01)
    for n in range(1000, -1, -10):
        led.duty_u16(n * 65)
        time.sleep(0.01)

🔍 对照着看

· 箱子里那句 led.bright(n)，拆开就是 duty_u16(n * 65)
· PWM(Pin(32), freq=1000) = 「32 号口，每秒开关 1000 次」

from box import LED
import time

r = LED(4)      # 红
g = LED(18)     # 绿

while True:
    for n in range(0, 1001, 10):   # 红灭绿亮
        r.bright(1000 - n)
        g.bright(n)
        time.sleep(0.01)
    for n in range(0, 1001, 10):   # 绿灭红亮
        g.bright(1000 - n)
        r.bright(n)
        time.sleep(0.01)
