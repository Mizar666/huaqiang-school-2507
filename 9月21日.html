# box.py - ESP32 极简教学封装库 (2507班实训全功能版)
# 适用平台: MicroPython (ESP32)
# 封装 LED、双灯、按键、触摸引脚、蜂鸣器旋律与 OLED 屏幕

import time
from machine import Pin, SoftI2C
import framebuf

# ==================== 1. LED 控制类 ====================
class LED:
    """LED 极简控制（默认 4 号引脚）"""
    def __init__(self, pin=4):
        self.pin_num = pin
        self.io = Pin(pin, Pin.OUT)
        self.off()

    def on(self):
        """点亮"""
        self.io.value(1)

    def off(self):
        """熄灭"""
        self.io.value(0)

    def toggle(self):
        """状态翻转"""
        self.io.value(not self.io.value())

    def blink(self, times=2, delay=0.15):
        """闪烁指定次数"""
        for _ in range(times):
            self.on()
            time.sleep(delay)
            self.off()
            time.sleep(delay)

# ==================== 2. 按键控制类 ====================
class Button:
    """按键控制（默认板载 BOOT 键 GPIO 0，按下返回 True）"""
    def __init__(self, pin=0):
        self.io = Pin(pin, Pin.IN, Pin.PULL_UP)

    def is_pressed(self):
        """判断是否按下"""
        return self.io.value() == 0

# ==================== 3. 触摸引脚类 ====================
class Touch:
    """电容触摸控制（默认 27 号引脚，手指碰触返回 True）"""
    def __init__(self, pin=27):
        from machine import TouchPad
        self.tp = TouchPad(Pin(pin))

    def read(self):
        """读取电容原始读数"""
        return self.tp.read()

    def is_touched(self, threshold=300):
        """判断手指是否触碰"""
        return self.tp.read() < threshold

# ==================== 4. 蜂鸣器与旋律 ====================
class Buzzer:
    """无源蜂鸣器（默认 23 号引脚，支持音调与简谱播放）"""
    def __init__(self, pin=23):
        self.pin = Pin(pin, Pin.OUT)

    def tone(self, freq, duration=0.15):
        """发声指定频率（Hz）与时长（秒）"""
        if freq <= 0:
            time.sleep(duration)
            return
        from machine import PWM
        p = PWM(self.pin, freq=int(freq), duty=512)
        time.sleep(duration)
        p.deinit()

    def beep(self, times=1, delay=0.1):
        """提示音哔哔声"""
        for _ in range(times):
            self.tone(1000, 0.08)
            time.sleep(delay)

    def play(self, notes="1231"):
        """播放数字简谱 (1~7 分别代表哆来咪发梭拉西，0为停顿)"""
        scale = {
            '1': 262, '2': 294, '3': 330, '4': 349,
            '5': 392, '6': 440, '7': 494, '0': 0
        }
        for note in str(notes):
            f = scale.get(note, 0)
            self.tone(f, 0.18)
            time.sleep(0.04)

# ==================== 5. OLED 屏幕类 (防呆容错) ====================
class _SSD1306(framebuf.FrameBuffer):
    def __init__(self, width, height, external_vcc):
        self.width = width
        self.height = height
        self.external_vcc = external_vcc
        self.pages = self.height // 8
        self.buffer = bytearray(self.pages * self.width)
        super().__init__(self.buffer, self.width, self.height, framebuf.MONO_VLSB)

    def show(self):
        self.write_cmd(0x21)
        self.write_cmd(0)
        self.write_cmd(self.width - 1)
        self.write_cmd(0x22)
        self.write_cmd(0)
        self.write_cmd(self.pages - 1)
        self.write_data(self.buffer)

class _SSD1306_I2C(_SSD1306):
    def __init__(self, width, height, i2c, addr=0x3C, external_vcc=False):
        self.i2c = i2c
        self.addr = addr
        self.temp = bytearray(2)
        self.write_list = [b"\x40", None]
        super().__init__(width, height, external_vcc)
        for cmd in (
            0xAE, 0xD5, 0x80, 0xA8, self.height - 1, 0xD3, 0x00,
            0x40, 0x8D, 0x14, 0x20, 0x00, 0xA1, 0xC8, 0xDA,
            0x12 if self.height == 64 else 0x02,
            0x81, 0xCF, 0xD9, 0xF1, 0xDB, 0x40, 0xA4, 0xA6, 0xAF
        ):
            self.write_cmd(cmd)
        self.fill(0)
        self.show()

    def write_cmd(self, cmd):
        self.temp[0] = 0x80
        self.temp[1] = cmd
        self.i2c.writeto(self.addr, self.temp)

    def write_data(self, buf):
        self.write_list[1] = buf
        self.i2c.writevto(self.addr, self.write_list)

class OLED:
    """OLED 屏幕（用法：oled = OLED(16, 17)；没接屏自动忽略、不打断其他实验）"""
    def __init__(self, scl=16, sda=17, width=128, height=64):
        self.ok = False
        self.screen = None
        try:
            self.i2c = SoftI2C(scl=Pin(scl), sda=Pin(sda), freq=100000, timeout=50000)
            devs = self.i2c.scan()
            addr = 0x3C if (0x3C in devs or 60 in devs) else (0x3D if (0x3D in devs or 61 in devs) else None)
            if addr:
                self.screen = _SSD1306_I2C(width, height, self.i2c, addr=addr)
                self.ok = True
                print("[OLED] 屏幕连接成功")
            else:
                print("[OLED 提示] 未检测到屏幕（不影响LED与其他实验运行）")
        except Exception:
            print("[OLED 提示] 屏幕暂未接入")

    def clear(self):
        if self.ok and self.screen:
            try:
                self.screen.fill(0)
                self.screen.show()
            except Exception:
                pass

    def display(self, line1="", line2="", line3="", line4=""):
        """一键全刷屏显示最多 4 行文字"""
        if self.ok and self.screen:
            try:
                self.screen.fill(0)
                for idx, content in enumerate([line1, line2, line3, line4]):
                    if content:
                        self.screen.text(str(content), 0, idx * 16, 1)
                self.screen.show()
            except Exception:
                pass

from box import LED
import time

# 初始化两个 LED 接口
led1 = LED(32)  # 黄灯：32 号口
led2 = LED(33)  # 红灯：33 号口

# 双灯交替闪烁
while True:
    led1.on()        # 1号灯点亮
    led2.off()       # 2号灯熄灭
    time.sleep(0.5)

    led1.off()       # 1号灯熄灭
    led2.on()        # 2号灯点亮
    time.sleep(0.5)

📋 参考接线对照

· 1号灯（黄）：32号口 ➔ 3b.i ➔ LED ➔ 4b.h 电阻 ➔ 负极轨
· 2号灯（红）：33号口 ➔ 11b.i ➔ LED ➔ 12b.h 电阻 ➔ 负极轨
· 公共地线：负极蓝线轨(-) ➔ 板子 GND
· 核心规律：两个独立控制通道，各自串联电阻，共用负极接地回路

参考答案 2 · 双灯触控互锁切换

仅供自查 · 对照互锁逻辑

Python 
from box import LED, Touch
import time

led1 = LED(32)     # 外接黄灯（32号口）
led2 = LED(33)     # 外接红灯（33号口）
touch = Touch(27)  # 触摸引脚（27号口，默认阈值300）

print("手摸亮灯实验开始！")

while True:
    if touch.is_touched():  # 触摸时返回 True
        led1.on()           # 黄灯亮
        led2.off()          # 红灯灭
    else:                   # 松手未触摸
        led1.off()          # 黄灯灭
        led2.on()           # 红灯亮

    time.sleep(0.05)

📋 互锁逻辑解析

· if 分支：手摸时，强制 led1.on() 并 led2.off()；
· else 分支：松手时，强制 led1.off() 并 led2.on()；
· 无需复杂 sleep 延时，纯由人体触摸实时决定两盏灯的互锁状态！

参考答案 3 · 课堂简答题要点

仅供核对 · 回路与元件原理

📝 简答 1：电流回路流向

· 电源流出：ESP32 32号引脚 输出高电平（3.3V）；
· 进入面包板：绿色跳线导入 3 列（3b.i ➔ 3b.f）；
· 流经小灯：从 LED 长脚（正极） 流入，短脚（负极） 流出至 4 列；
· 限流降压：经 4 列穿过 220Ω 电阻（4b.h ➔ 负极轨）；
· 闭合回到地：沿负极蓝线接地轨回到 ESP32 GND 引脚。

📝 简答 2：电阻的核心作用

· 核心功能：限流保护；
· 原因：LED 灯珠正向导通后自身内阻极小，相当于直通；
· 极端后果：如果不串联电阻，瞬时电流会远超引脚承受极限（ESP32 单引脚上限约 12~20mA）；
· 保护机制：串联 220Ω 电阻将工作电流限制在 安全区间（约 6~8mA），既保障发光，又防止烧毁 LED 与芯片。

【原理】面包板：谁跟谁通

看图认位置 · 再动手接线

三条铁律

· 数字同排：同一个数字下的 5 个孔本来就是一条线，插进 15a 就等于插进 15b
· 正负同列：红轨、蓝轨各自整条相通，随手挑一个孔都接得通
· 中间沟槽：上下两半互不相通，别指望跨过去

【原理】电阻：卡住电流的窄口

左右对比 · 明白 220Ω 在管什么

电阻在管什么

· LED 内阻几乎为零，直接接等于短路
· 220Ω 把电流压进 6~8mA 的安全区
· 拔掉电阻：灯珠和引脚一起受伤

【原理】为什么两行就能点灯

箱子里装的是什么 · 同一件事的两种写法

Python 
import time
from machine import Pin

io = Pin(32, Pin.OUT)   # 32号口设成输出口
io.value(1)             # 输出高电平，灯亮
time.sleep(0.5)
io.value(0)             # 输出低电平，灯灭

封装替你做了三件事

· from box import LED：把写好的 LED 类搬进来用
· LED(32)：__init__ 跑一遍，内部就是 Pin(32, Pin.OUT)
· led.on()：内部一句 io.value(1)，输出高电平
· led.off()：内部一句 io.value(0)，输出低电平
· OLED(16, 17)：内部 SoftI2C(scl=Pin(16), sda=Pin(17))，屏幕挂在 16、17 两个口上
